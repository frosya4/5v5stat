package main

import (
	"context"
	"fmt"
	"log"
	"time"

	firebase "firebase.google.com/go"
	"golang.org/x/crypto/bcrypt"
	"google.golang.org/api/option"
)

type User struct {
	ID           string    `json:"id" firestore:"id"`
	Email        string    `json:"email" firestore:"email"`
	PasswordHash string    `json:"-" firestore:"password_hash"`
	DisplayName  string    `json:"display_name" firestore:"display_name"`
	SteamID      string    `json:"steamid" firestore:"steamid"`
	Role         string    `json:"role" firestore:"role"` // "admin" or "user"
	CreatedAt    time.Time `json:"created_at" firestore:"created_at"`
}

func main() {
	ctx := context.Background()
	sa := option.WithCredentialsFile("serviceAccountKey.json")
	app, err := firebase.NewApp(ctx, nil, sa)
	if err != nil {
		log.Fatalf("Init error: %v\n", err)
	}
	client, err := app.Firestore(ctx)
	if err != nil {
		log.Fatalf("Firestore error: %v\n", err)
	}
	defer client.Close()

	email := "admin@admin.com"
	password := "admin123"
	
	// Check if admin already exists
	iter := client.Collection("users").Where("email", "==", email).Limit(1).Documents(ctx)
	doc, err := iter.Next()
	if err == nil && doc != nil {
		fmt.Println("Admin user already exists! Email:", email, "Password:", password)
		return
	}

	hash, _ := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)

	user := User{
		ID:           fmt.Sprintf("u_%d", time.Now().UnixNano()),
		Email:        email,
		PasswordHash: string(hash),
		DisplayName:  "Admin",
		SteamID:      "",
		Role:         "admin",
		CreatedAt:    time.Now(),
	}

	_, err = client.Collection("users").Doc(user.ID).Set(ctx, user)
	if err != nil {
		log.Fatalf("Failed to save user: %v", err)
	}
	fmt.Println("Admin user created! Email:", email, "Password:", password)
}
